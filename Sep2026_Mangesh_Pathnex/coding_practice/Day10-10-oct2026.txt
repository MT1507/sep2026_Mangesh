# Ansible -- Create a docker containers for multiple Applications

```yaml

- name: setup Docker container for multiple apps
  hosts: all
  become: yes
  tasks:
    - name: pull web app image
      docker_image:
        name: my-web-app
        source: pull

    - name: Pull Database app image
      docker_image:
        name: my-database-app
        source: pull

    - name: Run web app container
      docker_container:
        name: web-app
        image: my-web-app
        state: started
        published_ports:
            - "8080:80"

    - name: Run database app container
      docker_container:
        name: database-app
        image: my-database-app
        state: started
        published_ports:
         - "3306:3306"

### Kubernates -- Deploying Statefulset

apiVersion: apps/v1
kind: StatefulSet
metadata:
    name: pathnex-db
spec:
    serviceName: "pathnex-db"
    replicas: 3
    selector:
        matchLabels:
            app: pathnex-db
    template:
        metadata:
            labels:
                app: pathnex-db
        spec:
            containers:
                - name: db
                  image: mysql:5.7
                  volumeMounts:
                    - mountPath: /var/lib/mysql
                      name: mysql-storage
    volumeClaimTemplates:
        - metadata:
            name: mysql-storage
          spec:
            accessModes: ["ReadWriteOnce"]
            resources:
                requests:
                    storage: 1Gi

# Jenkinsfile  -- Build and deploy to Kubernates
pipeline {
    agent any
    environment {
        IMAGE_NAME = "pathnex-web-app"
    }
    stages {
        stage('Build Docker Image') {
            steps {
                script{
                    docker.build("${env.IMAGE_NAME}")
                }
            }
        }
        stage('Push Docker Image') {
            steps {
                script {
                   docker.withRegistry ('https://docker.io', 'docker-credentials') {
                    docker.image("${env.IMAGE_NAME}").push('latest')

                   }
                }
            }
        }
        stage ('Deploy to kubernates') {
            steps {
                script {
                    sh 'kubectl apply -f deployment.yaml'
                }
            }
        }       
    }
}
